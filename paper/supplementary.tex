\section*{S1. Theoretical Soundness Proofs for Two-Tier Core Extraction}

\begin{proposition}[Conservative One-Way Witness Extension]
\label{prop:witness}
Let $\mathcal{H}(\emptyset) = O_1 \cup \dots \cup O_k$ be an OWL~2~EL ontology union over signature $\Sigma_0$. Suppose for each anonymous class expression $E_\ell$ appearing as an operand of $\mathtt{ObjectIntersectionOf}(G_1,\dots,G_m)$ on the left-hand side of a class inclusion or equivalence axiom, we introduce a fresh atomic class $P_\ell = \textup{\texttt{urn:clasp:anon:}}\ell \notin \Sigma_0$ together with the single inclusion axiom $E_\ell \sqsubseteq P_\ell$, yielding $\mathcal{H}^+(\emptyset)$. Then for every pair of class expressions $C, D$ over $\Sigma_0$ (including $D = \bot$):
\[
  \mathcal{H}^+(\emptyset) \models C \sqsubseteq D \quad \Longleftrightarrow \quad \mathcal{H}(\emptyset) \models C \sqsubseteq D.
\]
\end{proposition}
\noindent\textit{Proof.} $(\Leftarrow)$ Immediate by monotonicity of description logic entailment since $\mathcal{H}(\emptyset) \subseteq \mathcal{H}^+(\emptyset)$. $(\Rightarrow)$ Suppose $\mathcal{H}(\emptyset) \not\models C \sqsubseteq D$. Then there exists an interpretation $\mathcal{I}$ over domain $\Delta$ with extension map $\mathrm{ext}_{\mathcal{I}}(\cdot)$ such that $\mathcal{I} \models \mathcal{H}(\emptyset)$ and $\mathrm{ext}_{\mathcal{I}}(C) \not\subseteq \mathrm{ext}_{\mathcal{I}}(D)$. Define an extended interpretation $\mathcal{J}$ over $\Sigma_0 \cup \{P_\ell\}$ with the same domain $\Delta$ by setting $\mathrm{ext}_{\mathcal{J}}(X) = \mathrm{ext}_{\mathcal{I}}(X)$ for all $X \in \Sigma_0$ and $\mathrm{ext}_{\mathcal{J}}(P_\ell) = \mathrm{ext}_{\mathcal{I}}(E_\ell)$ for every fresh witness $P_\ell$. Because no $P_\ell$ occurs inside $E_\ell$ or inside $\mathcal{H}(\emptyset)$, structural induction gives $\mathrm{ext}_{\mathcal{J}}(E_\ell) = \mathrm{ext}_{\mathcal{I}}(E_\ell) = \mathrm{ext}_{\mathcal{J}}(P_\ell)$, so $\mathcal{J} \models E_\ell \sqsubseteq P_\ell$ for all $\ell$ and $\mathcal{J} \models \mathcal{H}(\emptyset)$. Hence $\mathcal{J} \models \mathcal{H}^+(\emptyset)$ while $\mathrm{ext}_{\mathcal{J}}(C) = \mathrm{ext}_{\mathcal{I}}(C) \not\subseteq \mathrm{ext}_{\mathcal{I}}(D) = \mathrm{ext}_{\mathcal{J}}(D)$, proving $\mathcal{H}^+(\emptyset) \not\models C \sqsubseteq D$. \hfill $\square$

\begin{proposition}[Property-Filtered $\bot$-Locality Trigger Soundness]
\label{prop:locality}
Let $\mathrm{isLocal}_{\bot}(\alpha, \Sigma)$ denote the OWL~2 syntactic $\bot$-locality evaluator~\cite{cuencagrau2008modular}, and let $\mathcal{P}$ be the set of all object and data properties in $\mathcal{H}$. If $\mathrm{isLocal}_{\bot}(\alpha, \mathcal{P}) = \mathrm{true}$, then for any signature $\Sigma$ containing zero classes and zero individuals from $\mathrm{Sig}(\alpha)$ ($\Sigma \cap (\mathrm{Sig}(\alpha) \setminus \mathcal{P}) = \emptyset$), $\mathrm{isLocal}_{\bot}(\alpha, \Sigma) = \mathrm{true}$.
\end{proposition}
\noindent\textit{Proof.} In syntactic $\bot$-locality, the grammar sets of $\bot$-equivalent expressions $\mathcal{C}^{\bot}(\Sigma)$ and $\top$-equivalent expressions $\mathcal{C}^{\top}(\Sigma)$ shrink monotonically with respect to signature inclusion: $\Sigma_1 \subseteq \Sigma_2 \implies \mathcal{C}^{\bot}(\Sigma_2) \subseteq \mathcal{C}^{\bot}(\Sigma_1)$ and $\mathcal{C}^{\top}(\Sigma_2) \subseteq \mathcal{C}^{\top}(\Sigma_1)$, whence $\neg \mathrm{isLocal}_{\bot}(\alpha, \Sigma_1) \implies \neg \mathrm{isLocal}_{\bot}(\alpha, \Sigma_2)$. Moreover, locality depends only on $\Sigma \cap \mathrm{Sig}(\alpha)$. If $\Sigma \cap (\mathrm{Sig}(\alpha) \setminus \mathcal{P}) = \emptyset$, then $\Sigma \cap \mathrm{Sig}(\alpha) \subseteq \mathcal{P}$. Since $\mathrm{isLocal}_{\bot}(\alpha, \mathcal{P}) = \mathrm{true}$, monotonicity yields $\mathrm{isLocal}_{\bot}(\alpha, \Sigma \cap \mathrm{Sig}(\alpha)) = \mathrm{isLocal}_{\bot}(\alpha, \Sigma) = \mathrm{true}$. Consequently, $\alpha$ need only be indexed under trigger entities in $\mathrm{Sig}(\alpha) \setminus \mathcal{P}$. \hfill $\square$
